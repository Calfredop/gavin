# The repo's CI, and the thing that keeps the ports green. Every developer
# here is on a mac, so macOS is the platform that gets exercised by hand a
# hundred times a day and the other two are the ones that would silently
# rot between releases.
#
# The two jobs are deliberately at different stages. Linux is DONE: its
# suites are green and every step is a gate. Windows is a port in flight,
# so only the thing that has been made to work is a gate -- that the three
# Rust crates COMPILE for the target -- and everything past it reports
# without failing the build, until the verification pass on a real Windows
# machine turns each one into a gate.
name: CI

on:
  push:
    branches: [main]
  pull_request:
  workflow_dispatch:

# A second push to the same branch makes the first run's answer stale.
concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

jobs:
  linux:
    name: Linux (Rust + app)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      # `app/src-tauri` is a workspace member, so `cargo test --workspace`
      # links against the whole WebKitGTK stack even though nothing here
      # opens a window. This is the Tauri v2 Ubuntu list.
      - name: Install the GTK/WebKit stack tauri links against
        run: |
          sudo apt-get update
          sudo apt-get install --no-install-recommends -y \
            libwebkit2gtk-4.1-dev \
            build-essential \
            curl \
            wget \
            file \
            libxdo-dev \
            libssl-dev \
            libayatana-appindicator3-dev \
            librsvg2-dev

      - uses: dtolnay/rust-toolchain@stable
      - uses: Swatinem/rust-cache@v2

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: app/package-lock.json

      - name: npm ci
        working-directory: app
        run: npm ci

      # The daemon's fs-watcher module first and by itself. It is flaky
      # under full-suite cargo parallelism -- watcher timing, not the code
      # (CLAUDE.md says to re-run it alone before calling a failure a
      # regression). Running it alone up front and excluding it from the
      # sweep removes the CAUSE rather than retrying after the symptom, so
      # a red build below means something is actually broken.
      - name: cargo test — the fs-watcher module, alone
        run: cargo test -p gavin-daemon gavin:: -- --test-threads=1

      - name: cargo test --workspace
        run: cargo test --workspace -- --skip gavin::

      - name: npm test
        working-directory: app
        run: npm test

      - name: npm run check
        working-directory: app
        run: npm run check

      - name: npm run build
        working-directory: app
        run: npm run build

  windows:
    name: Windows (compiles)
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4

      - uses: dtolnay/rust-toolchain@stable
      - uses: Swatinem/rust-cache@v2

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: app/package-lock.json

      - name: npm ci
        working-directory: app
        run: npm ci

      # THE GATE. `std::os::unix` does not exist on this target, so before
      # the windows-port card these three crates did not fail to run on
      # Windows -- they failed to parse. Keeping that answer honest is what
      # this job is for, and it is the one step here allowed to be red.
      #
      # These three and not `--workspace`: app/src-tauri drags in the whole
      # WebView2 and tauri-build chain, which is a different question with a
      # different failure mode. It is checked below, where a failure reports
      # instead of blocking.
      - name: cargo check — the daemon, the MCP server and the protocol
        run: cargo check -p protocol -p gavin-daemon -p gavin-mcp

      # Everything below reports. A red step here is a piece of the port
      # that is not finished; promote it to a gate when it goes green.
      - name: cargo check --workspace --tests
        continue-on-error: true
        run: cargo check --workspace --tests

      # Same carve-out as the Linux job: the fs-watcher module is flaky
      # under full-suite parallelism (watcher timing, not the code), so it
      # runs alone and is skipped in the sweep.
      - name: cargo test — the fs-watcher module, alone
        continue-on-error: true
        run: cargo test -p gavin-daemon gavin:: -- --test-threads=1

      - name: cargo test --workspace
        continue-on-error: true
        run: cargo test --workspace -- --skip gavin::

      - name: npm test
        continue-on-error: true
        working-directory: app
        run: npm test

      - name: npm run check
        continue-on-error: true
        working-directory: app
        run: npm run check

      - name: npm run build
        continue-on-error: true
        working-directory: app
        run: npm run build
